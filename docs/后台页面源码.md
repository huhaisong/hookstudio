## 1. web/version.js —— 版本闸（最先加载）

服务器换了新版，旧页面必须刷新后才能继续用。比对这个页面的 <meta name=app-build>、/api/status 的 build、每个响应的 X-App-Build 头，任一对不上就全屏锁屏，只留一个「立即刷新」。


```js
/* 版本闸：服务器换了新版，旧页面必须刷新后才能继续用。
 *
 * 为什么必须有这个东西（2026-11-04 用户要求「需要更新时必须更新之后才可继续使用」）：
 *   后端改了接口，浏览器里还跑着上一次的 app.js —— 按钮点了没反应、参数按旧格式
 *   提交、AI 授权票据不认，用户只会觉得「坏了」。这类问题排查成本极高，
 *   而根因只有一句话：页面和服务器不是同一版。靠人记着按 Ctrl+F5 不可靠。
 *
 * 三处指纹，任意两处不一致就锁屏：
 *   ① 页面里的 <meta name="app-build">（服务端渲染 HTML 时注入，代表「这个页面是哪一版」）
 *   ② /api/status 响应的 build（代表「服务器现在是哪一版」）
 *   ③ 每个响应头 X-App-Build（请求途中后端被更新也能立刻发现）
 *
 * 锁屏是硬的：全屏遮罩 + 只有一个「立即刷新」。不做「稍后再说」——那就是把旧页面
 * 又放回去用了，等于没做。
 */
(function (global) {
  const meta = document.querySelector('meta[name="app-build"]');
  const BUILD = (meta && meta.content) || "";
  const K_TRY = "hs_upd_try";

  let locked = false;
  let mask = null;

  function esc(s) {
    return String(s == null ? "" : s).replace(/[&<>"]/g, (c) => (
      { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
  }

  function tries() {
    try { return Number(sessionStorage.getItem(K_TRY) || 0); } catch (e) { return 0; }
  }

  function bumpTries() {
    try { sessionStorage.setItem(K_TRY, String(tries() + 1)); } catch (e) { /* 无痕模式 */ }
  }

  /** 锁屏。serverB 只用于显示，不参与判断。 */
  function lock(serverB) {
    if (locked) return;
    locked = true;
    document.documentElement.classList.add("upd-lock");

    const n = tries();
    bumpTries();
    // 刷了一次还对不上，说明浏览器或中间层还留着旧东西（代理/CDN/强缓存），
    // 这时只给「刷新」按钮就是死循环，得把话说清楚。
    const stuck = n >= 2;

    mask = document.createElement("div");
    mask.className = "upd-mask";
    mask.innerHTML = `
      <div class="upd-card">
        <div class="upd-ico">⟳</div>
        <h1>系统已更新，需要刷新后继续</h1>
        <p class="upd-lead">这个页面是旧版本，服务器已经换成新版本。
          继续操作可能按旧格式保存（出片参数、AI 授权、日志都会对不上），
          所以已暂停使用。</p>
        <div class="upd-code">
          <span>页面 <b>${esc(BUILD || "未知")}</b></span>
          <span>服务器 <b>${esc(serverB || "未知")}</b></span>
        </div>
        <button class="upd-btn" id="upd-reload">立即刷新</button>
        <p class="upd-tip">${stuck
          ? "刷新后仍提示？按 <b>Ctrl + F5</b>（Mac：⌘ + Shift + R）强制重新加载。"
          : "刷新后会自动回到当前页面，不用重新登录。"}</p>
      </div>`;
    document.body.appendChild(mask);
    const btn = mask.querySelector("#upd-reload");
    btn.addEventListener("click", () => {
      btn.disabled = true;
      btn.textContent = "正在刷新…";
      // 清掉「试过几次」，否则下次正常更新也会被当成刷不干净
      setTimeout(() => { try { sessionStorage.removeItem(K_TRY); } catch (e) {} }, 4000);
      location.reload();
    });
    // 焦点拉过来，避免用户还在键盘上操作后面的表单
    setTimeout(() => btn.focus(), 30);
  }

  /** 比指纹锁更硬的一种锁：这版程序本身不能用了，刷新解决不了，必须换安装包。
   *
   *  什么时候会走到这里：服务器把「最低可用版本」调到比本机高（见 app/update.py）。
   *  exe 分发给别人之后，本地指纹闸永远不会触发——用户电脑上的页面和后端是同一份
   *  安装包，两者天然一致。只有服务器说了算才能实现「必须更新后才能继续使用」。
   */
  function lockUpgrade(info) {
    if (locked) return;
    locked = true;
    document.documentElement.classList.add("upd-lock");
    info = info || {};
    const local = info.local || "";
    const need = info.latest || info.min || "";
    const url = info.url || "";
    mask = document.createElement("div");
    mask.className = "upd-mask";
    mask.innerHTML = `
      <div class="upd-card upd-hard">
        <div class="upd-ico">⤓</div>
        <h1>这个版本已停止使用</h1>
        <p class="upd-lead">服务器已经不再支持当前版本，为避免参数错位、出片失败、
          数据对不上，已暂停使用。请安装新版后继续，<b>刷新页面不起作用</b>。</p>
        <div class="upd-code">
          <span>当前版本 <b>${esc(local || "未知")}</b></span>
          <span>需要版本 <b>${esc(need || "更新")}</b></span>
        </div>
        ${info.note ? `<p class="upd-note">${esc(info.note)}</p>` : ""}
        ${url
          ? `<a class="upd-btn" href="${esc(url)}" target="_blank" rel="noopener">下载新版</a>`
          : `<button class="upd-btn" disabled>请联系管理员获取新版安装包</button>`}
        <p class="upd-tip">装好之后关掉黑窗口再双击新的 钩子工坊.exe，浏览器会自动打开；
          原来的项目和成片都在，不会丢。</p>
      </div>`;
    document.body.appendChild(mask);
  }

  /** 比指纹。serverB 只用于显示，不参与判断。 */
  function check(serverB) {
    if (!BUILD || !serverB) return false;
    if (serverB === BUILD) {
      // 恰好一致：说明这次刷新是干净的，把计数清掉
      try { sessionStorage.removeItem(K_TRY); } catch (e) { /* ignore */ }
      return false;
    }
    lock(serverB);
    return true;
  }

  // ── 包装 fetch：任何响应都能发现「后端换版了」／「这版被封了」──────────
  // 只加响应头判断，不改变请求行为；锁屏后照常放行，只是界面已经拦住用户了。
  const rawFetch = global.fetch.bind(global);
  global.fetch = function (input, init) {
    return rawFetch(input, init).then((res) => {
      // 426 = 服务器要求升级。详情在 body 里，读一份克隆，别把下游要读的流吃掉。
      if (res.status === 426) {
        res.clone().json().then((d) => lockUpgrade(d)).catch(() => lockUpgrade({}));
        return res;
      }
      const b = res.headers && res.headers.get("X-App-Build");
      if (b) check(b);
      return res;
    });
  };

  /** 主动向服务端问一次当前版本（含「服务器要求的最低版本」）。 */
  function poll() {
    if (locked || document.hidden) return Promise.resolve(false);
    return rawFetch("/api/status", { cache: "no-store" })
      .then((r) => r.json())
      .then((d) => {
        // 远程闸优先：本地指纹都懒比了，这版已经不该用
        if (d && d.update && d.update.need) {
          lockUpgrade(Object.assign({ local: d.version }, d.update.need));
          return true;
        }
        return check(d && d.build);
      })
      .catch(() => false);
  }

  // 页面放着不动也要能发现（很多人是开着窗口去泡咖啡，回来接着点）
  setInterval(poll, 20000);
  setTimeout(poll, 2500);

  global.AppVersion = { build: BUILD, check, lock, lockUpgrade, poll, locked: () => locked };
})(window);
```

## 3. web/admin.js —— 管理后台逻辑

用户表格渲染与操作、日志筛选/展开/导出、设置读写、权限自检。**所有插入 innerHTML 的字段都过 esc()**——用户名、封禁理由、日志 path 都是用户可控内容。


```js
/* 管理后台：用户管理 / 操作日志 / 系统设置。
 *
 * 全部走 Session.ureq（自动带 token、401 自动刷新）。
 * 渲染一律经过 esc()——用户名、备注、日志里的 path 都可能是用户可控的内容，
 * 直接拼进 innerHTML 等于把后台拱手送人。
 */
const $ = (s) => document.querySelector(s);
const esc = Session.esc;
const ftime = Session.ftime;
const ROLE_CN = Session.ROLE_CN;
const STATUS_CN = Session.STATUS_CN;

const ST = {
  page: { users: 1, logs: 1 },
  pages: { users: 1, logs: 1 },
  me: null,
  action: null,   // 弹窗回调
};

/* ── 通用：toast / 弹窗 / 分页 ─────────────────────────────────────── */
let toastTimer = null;
function toast(text) {
  let el = $("#ad-toast");
  if (!el) {
    el = document.createElement("div");
    el.id = "ad-toast";
    el.className = "ad-toast";
    document.body.appendChild(el);
  }
  el.textContent = text;
  el.classList.add("on");
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => el.classList.remove("on"), 2800);
}

/** 统一的输入弹窗（封禁要填理由，原生 prompt 在某些浏览器被拦） */
function ask(title, desc, { value = "", placeholder = "", withInput = true } = {}) {
  return new Promise((resolve) => {
    $("#mo-title").textContent = title;
    $("#mo-desc").textContent = desc || "";
    const inp = $("#mo-input");
    inp.style.display = withInput ? "" : "none";
    inp.value = value;
    inp.placeholder = placeholder;
    $("#ad-modal").classList.remove("hidden");
    if (withInput) setTimeout(() => inp.focus(), 30);
    ST.action = () => {
      $("#ad-modal").classList.add("hidden");
      ST.action = null;
      resolve(withInput ? inp.value.trim() : true);
    };
    $("#mo-ok").onclick = () => ST.action && ST.action();
    $("#mo-cancel").onclick = () => {
      $("#ad-modal").classList.add("hidden");
      ST.action = null;
      resolve(null);
    };
  });
}

function pageBar(el, page, pages, onGo) {
  el.innerHTML = `<button class="ghost sm" ${page <= 1 ? "disabled" : ""} data-go="${page - 1}">上一页</button>
    <span class="hint">第 ${page} / ${Math.max(1, pages)} 页</span>
    <button class="ghost sm" ${page >= pages ? "disabled" : ""} data-go="${page + 1}">下一页</button>`;
  el.querySelectorAll("button[data-go]").forEach((b) => {
    b.addEventListener("click", () => onGo(Number(b.dataset.go)));
  });
}

/* ── 用户管理 ───────────────────────────────────────────────────────── */
async function loadUsers(page = 1) {
  ST.page.users = page;
  const q = $("#u-q").value.trim();
  const role = $("#u-role").value;
  const status = $("#u-status").value;
  const url = `/api/admin/users?page=${page}&size=20&q=${encodeURIComponent(q)}`
    + `&role=${role}&status=${status}`;
  let data;
  try {
    data = await Session.ureq(url);
  } catch (e) {
    toast("加载用户失败：" + e.message);
    return;
  }
  ST.pages.users = data.pages;
  const c = data.counts || {};
  $("#u-stat").textContent = `共 ${c.total || 0} 人 · 管理员 ${c.admins || 0} · 已封禁 ${c.banned || 0}`;

  $("#u-rows").innerHTML = (data.items || []).map((u) => {
    const banned = u.status === "banned";
    const self = ST.me && u.id === ST.me.id;
    const isAdmin = u.role === "admin";
    return `<tr class="${banned ? "ad-banned" : ""}">
      <td class="ad-mono">${u.id}</td>
      <td>
        <div class="ad-user">
          <span class="ad-avatar">${esc((u.display_name || u.username).slice(0, 1))}</span>
          <div>
            <b>${esc(u.display_name || u.username)}</b>
            <span class="ad-handle">@${esc(u.username)}</span>
            ${self ? '<span class="ad-tag">我</span>' : ""}
            ${u.must_change_pwd ? '<span class="ad-tag warn">初始密码</span>' : ""}
            ${banned && u.ban_reason ? `<div class="ad-ban-reason">封禁理由：${esc(u.ban_reason)}</div>` : ""}
          </div>
        </div>
      </td>
      <td class="ad-dim">${esc(u.email || "—")}</td>
      <td><span class="ad-tag ${isAdmin ? "admin" : ""}">${ROLE_CN[u.role] || u.role}</span></td>
      <td><span class="ad-tag ${banned ? "bad" : "ok"}">${STATUS_CN[u.status] || u.status}</span></td>
      <td class="ad-dim">${ftime(u.last_login_at)}</td>
      <td class="ad-dim">${ftime(u.last_active)}</td>
      <td class="ad-ops">
```

## 4. web/session.js —— 会话工具（三个页面共用）

存取令牌、带凭证请求、401 自动刷新（单飞）、跳登录。原生 JS，不引框架。


```js
/* 会话工具：登录页 / 管理后台 / 主界面共用。
 *
 * 只做四件事：存取令牌、带凭证请求、401 自动刷新、跳登录。
 * 不引框架——这个项目的前端就是原生 JS，多一个依赖就多一份打包风险。
 *
 * 令牌存在 localStorage（不是 cookie）：本机工具没有 CSRF 面，
 * localStorage 更好调试；如果要部署到公网做多租户，建议改成
 * HttpOnly + SameSite=Lax 的 cookie（那时刷新逻辑要跟着改）。
 */
(function (global) {
  const K_ACCESS = "hs_access";
  const K_REFRESH = "hs_refresh";
  const K_USER = "hs_user";

  const S = {
    get access() { return localStorage.getItem(K_ACCESS) || ""; },
    get refresh() { return localStorage.getItem(K_REFRESH) || ""; },
    get user() {
      try { return JSON.parse(localStorage.getItem(K_USER) || "null"); } catch (e) { return null; }
    },
    save(data) {
      if (data.access_token) localStorage.setItem(K_ACCESS, data.access_token);
      if (data.refresh_token) localStorage.setItem(K_REFRESH, data.refresh_token);
      if (data.user) localStorage.setItem(K_USER, JSON.stringify(data.user));
    },
    setUser(u) { localStorage.setItem(K_USER, JSON.stringify(u || null)); },
    clear() {
      [K_ACCESS, K_REFRESH, K_USER].forEach((k) => localStorage.removeItem(k));
    },
    /** 跳登录页，带回跳地址和「为什么被踢出来」（登录页要按这个给用户一句话） */
    toLogin(next, reason) {
      const to = next || (location.pathname + location.search);
      if (location.pathname === "/login") return;
      let url = "/login?next=" + encodeURIComponent(to);
      if (reason) url += "&reason=" + encodeURIComponent(reason);
      location.href = url;
    },
  };

  // 刷新单飞：同一时刻只允许一个刷新请求在跑。
  // 服务端做了「轮换」（旧 refresh 用一次就作废），并发刷新会让后到的那次
  // 拿着已作废的 token 直接失败，用户莫名其妙掉线。
  let refreshing = null;

  async function doRefresh() {
    if (!S.refresh) return false;
    if (refreshing) return refreshing;
    refreshing = (async () => {
      try {
        const r = await fetch("/api/auth/refresh", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ refresh_token: S.refresh }),
        });
        if (!r.ok) { S.clear(); return false; }
        const data = await r.json();
        S.save(data);
        return true;
      } catch (e) {
        return false;
      } finally {
        refreshing = null;
      }
    })();
    return refreshing;
  }

  /** 带凭证请求：自动加 Bearer，401 时刷新一次再重试。
   *
   * 401 / 403 的分工（2026-11-04 补）：
   *   401 = 没凭证或凭证过期 → 试一次刷新，失败就送登录页（reason=expired）
   *   403 + code=banned = 账号被封 → 清凭证、送登录页（reason=banned）
   *   403 其它 = 权限不够（普通用户点管理员接口）→ **留在原地**，只报错。
   * 把 403 一律当掉线处理会让普通用户莫名其妙被踢到登录页。
   */
  async function ureq(url, opts = {}) {
    const headers = Object.assign(
      { "Content-Type": "application/json" },
      opts.headers || {},
      S.access ? { Authorization: "Bearer " + S.access } : {}
    );
    const r = await fetch(url, Object.assign({}, opts, { headers }));
    if (r.status === 401 && S.refresh) {
      const ok = await doRefresh();
      if (ok) return ureq(url, opts);
      S.clear();
    }
    const txt = await r.text();
    let data;
    try { data = JSON.parse(txt); } catch (e) { data = { detail: txt.slice(0, 300) }; }
    if (!r.ok) {
      if (data && data.code === "banned") {
        S.clear();
        S.toLogin(null, "banned");
      } else if (data && data.code === "must-change-pwd") {
        // 不是掉线、也不是没权限，是「还没更新密码」——把人按在改密弹窗里
        openPwdGate(data.detail);
      } else if (r.status === 401) {
        S.clear();
        // revoked = 被踢下线 / 改过密码；expired = 单纯过期
        S.toLogin(null, data && data.code === "revoked" ? "kicked" : "expired");
      }
      const err = new Error(data.detail || ("HTTP " + r.status));
      err.status = r.status;
      err.data = data;
      throw err;
    }
    return data;
  }

  function esc(s) {
    return String(s == null ? "" : s).replace(/[&<>"]/g, (c) => (
      { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
  }

  /* ── 强制改密闸（2026-11-04）─────────────────────────────────────────
   * 后端在中间件里硬拦：`must_change_pwd=1` 的用户除了「读自己是谁 / 改密 / 退出 /
   * 续期」之外，所有接口都是 403 + code=must-change-pwd。
   * 这里负责把人接住：弹一个**关不掉**的卡片，改完才继续。
   *
   * 为什么不只是 toast 提示：提示可以被无视（关标签页、直接 curl），
   * 用户要的是「更新之后才可继续使用」，那就得让界面也走不过去。
   */
  let gateOpen = false;

  function openPwdGate(msg) {
    if (gateOpen) return;
    if (location.pathname === "/login") return;   // 登录页自己引导
    gateOpen = true;
    const mask = document.createElement("div");
    mask.className = "gp-mask";
    mask.innerHTML = `
      <div class="gp-card">
        <h1>请先设置新密码</h1>
        <p class="gp-lead">${esc(msg || "你还在使用初始密码（或管理员刚重置的密码）。改完之后才能继续使用。")}</p>
        <label class="gp-fld">原密码
          <input id="gp-old" type="password" autocomplete="current-password" placeholder="当前正在用的密码">
        </label>
        <label class="gp-fld">新密码
          <input id="gp-new" type="password" autocomplete="new-password" placeholder="至少 8 位，不能是纯数字">
        </label>
        <label class="gp-fld">确认新密码
          <input id="gp-new2" type="password" autocomplete="new-password" placeholder="再输一次">
        </label>
        <p class="gp-err" id="gp-err"></p>
        <button class="gp-btn" id="gp-save">保存并继续</button>
        <p class="gp-tip">改完后其他设备会退出登录，这台电脑继续用。</p>
      </div>`;
    document.body.appendChild(mask);
    const err = mask.querySelector("#gp-err");
    const btn = mask.querySelector("#gp-save");
    const iOld = mask.querySelector("#gp-old");
    const iNew = mask.querySelector("#gp-new");
    const iNew2 = mask.querySelector("#gp-new2");
    setTimeout(() => iOld.focus(), 50);

    async function submit() {
      err.textContent = "";
      const oldPwd = iOld.value;
      const newPwd = iNew.value;
      if (!oldPwd) { err.textContent = "请输入原密码"; iOld.focus(); return; }
      if (newPwd.length < 8) { err.textContent = "新密码至少 8 位"; iNew.focus(); return; }
      if (newPwd !== iNew2.value) { err.textContent = "两次输入的新密码不一样"; iNew2.focus(); return; }
      btn.disabled = true;
      btn.textContent = "保存中…";
      try {
        const r = await fetch("/api/auth/password", {
          method: "PUT",
          headers: Object.assign({ "Content-Type": "application/json" },
            S.access ? { Authorization: "Bearer " + S.access } : {}),
          body: JSON.stringify({ old_password: oldPwd, new_password: newPwd }),
        });
        const d = await r.json().catch(() => ({}));
        if (!r.ok) {
          err.textContent = d.detail || ("HTTP " + r.status);
          btn.disabled = false;
          btn.textContent = "保存并继续";
          return;
        }
        // 改密会让所有旧 token 作废，后端给当前设备发了一对新凭证
        if (d && d.access_token) S.save(d);
        btn.textContent = "已保存，正在重新加载…";
        // 整页重载最稳：所有模块都会按「已改密」的新身份重新拉一遍状态
        setTimeout(() => location.reload(), 300);
      } catch (e) {
        err.textContent = "连不上服务：" + e.message;
        btn.disabled = false;
        btn.textContent = "保存并继续";
      }
    }

    btn.addEventListener("click", submit);
    mask.querySelectorAll("input").forEach((el) => {
      el.addEventListener("keydown", (e) => { if (e.key === "Enter") submit(); });
    });
  }

  /** ISO(UTC) → 本地 `11-04 16:20:31`。库里全存 UTC，显示时才转。 */
  function ftime(iso) {
    if (!iso) return "—";
    const d = new Date(iso);
    if (isNaN(d.getTime())) return String(iso).slice(0, 19);
    const p = (n) => String(n).padStart(2, "0");
    return `${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
  }

  const ROLE_CN = { admin: "管理员", user: "普通用户" };
  const STATUS_CN = { active: "正常", banned: "已封禁" };

  global.Session = {
    state: S,
    ureq,
    esc,
    ftime,
    doRefresh,
    /** 前端主动弹改密闸（登录返回 must_change_pwd=1 时用） */
    requirePasswordChange: openPwdGate,
    passwordGateOpen: () => gateOpen,
    ROLE_CN,
    STATUS_CN,
    logout(allDevices) {
      const rt = S.refresh;
      return fetch("/api/auth/logout", {
        method: "POST",
        headers: Object.assign({ "Content-Type": "application/json" },
          S.access ? { Authorization: "Bearer " + S.access } : {}),
        body: JSON.stringify({ refresh_token: rt, all_devices: !!allDevices }),
      }).catch(() => {}).then(() => S.clear());
    },
  };
})(window);
```
